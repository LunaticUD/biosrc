# FastQC质控
## fastp
1. 单文件运行
	```bash
	#!/bin/bash
	# run_fastp_single.sh
	# 定义输入文件和输出文件前缀
	input_forward="sample_1.fastq.gz"
	input_reverse="sample_2.fastq.gz"
	output_prefix="clean_"
	# 运行fastp
	fastp -i "$input_forward" -I "$input_reverse" \
      	-o "${output_prefix}${input_forward}" \
      	-O "${output_prefix}${input_reverse}" \
      	-h "${output_prefix}fastp_report.html" \
      	-j "${output_prefix}fastp_report.json" \
      	--thread 4
	# 使用4个CPU线程加速
	echo "Fastp质控完成！报告见 ${output_prefix}fastp_report.html"
	```
2. 多文件运行
	```bash
	# !/bin/bash
	# run_fastp_parallel.sh
	# 文件名命名规则一致：*_1.fastq.gz 和 *_2.fastq.gz
	for forward_file in *_1.fastq.gz
	do
		base_name=${forward_file%_1.fastq.gz}
		reverse_file="${base_name}_2.fastq.gz"
		output_prefix="${base_name}_clean_"
		echo "处理样本：$base_name"
		fastp -i "$forward_file" -I "$reverse_file" \
	          -o "${output_prefix}1.fastq.gz" \
	          -O "${output_prefix}2.fastq.gz" \
	          -h "${output_prefix}fastp_report.html" \
	          -j "${output_prefix}fastp_report.json" \
	          --thread 2 &          
	          # 后台运行，每个任务用2个线程
	done
	wait
	echo "所有样本Fastp质控已完成！"
	```
## Fastqc+trimmomatic
1. 单文件运行
	```bash
	#!/bin/bash
	# run_trimmomatic_single.sh
	sample_name="sample"
	input_forward="${sample_name}_1.fastq.gz"
	input_reverse="${sample_name}_2.fastq.gz"
	# Step 1: 使用FastQC进行质控前评估
	echo "Step 1: 运行FastQC进行原始数据质量评估..."
	fastqc "$input_forward" "$input_reverse" -t 2 # -t 指定线程数
	# Step 2: 使用Trimmomatic进行质控
	echo "Step 2: 运行Trimmomatic进行数据质控..."
	trimmomatic PE -threads 4 \
    	"$input_forward" "$input_reverse" \
    	"${sample_name}_clean_1.fastq.gz" "${sample_name}_unpaired_1.fastq.gz" \
    	"${sample_name}_clean_2.fastq.gz" "${sample_name}_unpaired_2.fastq.gz" \
    	ILLUMINACLIP:TruSeq3-PE-2.fa:2:30:10:2:keepBothReads \
    	LEADING:3 TRAILING:3 SLIDINGWINDOW:4:20 MINLEN:50    
    	# 参数解释：    
    	# PE: 双端数据    
    	# ILLUMINACLIP: 切除接头。TruSeq3-PE-2.fa是适配器序列文件。    
    	# LEADING:3 从 reads 的开头切除质量值低于3的碱基    
    	# TRAILING:3 从 reads 的末尾切除质量值低于3的碱基    
    	# SLIDINGWINDOW:4:20 使用4碱基宽的滑动窗口，当窗口内平均质量值低于20时，切除其后所有序列    
    	# MINLEN:50 丢弃长度小于50bp的reads
	echo "Trimmomatic质控完成！"
	echo "高质量配对数据：${sample_name}_clean_1.fastq.gz 和 ${sample_name}_clean_2.fastq.gz"
	echo "未配对的数据被单独存放。"
	```
2. 多文件运行
	```bash
	#!/bin/bash
	# run_trimmomatic_parallel.sh
	# 先批量运行FastQC
	echo "批量运行FastQC..."
	fastqc *_1.fastq.gz *_2.fastq.gz -t 8
	# 然后批量运行Trimmomatic
	for forward_file in *_1.fastq.gz
	do    
		base_name=${forward_file%_1.fastq.gz}    
		reverse_file="${base_name}_2.fastq.gz"    
		echo "正在用Trimmomatic处理样本：$base_name ..."
		trimmomatic PE -threads 2 \
			"$input_forward" "$input_reverse" \
			"${sample_name}_clean_1.fastq.gz" "${sample_name}_unpaired_1.fastq.gz" \
			"${sample_name}_clean_2.fastq.gz" "${sample_name}_unpaired_2.fastq.gz" \
			ILLUMINACLIP:TruSeq3-PE-2.fa:2:30:10:2:keepBothReads \
			LEADING:3 TRAILING:3 SLIDINGWINDOW:4:20 MINLEN:50 &
	done
	wait
	echo "所有样本的Trimmomatic质控已完成！"
	```

	```bash 
	# run_kneaddata.sh
	db_path="db"
	for forward_file in *_1.fastq.gz
	do    
		base_name=${forward_file%_1.fastq.gz}    
		reverse_file="${base_name}_2.fastq.gz"    
		output_dir="kneaddata_output_${base_name}" 
		# 为每个样本创建独立的输出目录    
		echo "正在用KneadData处理样本：$base_name ..."    
		kneaddata -i1 “$forward_file” -i2 "$reverse_file" \
			-o $output_dir \
			-db "$db_path/Homo_sapiens_Bowtie2_v1" \
			-t 2 \
			--trimmomatic-options "ILLUMINACLIP:TruSeq3-PE-2.fa:2:30:10:2:keepBothReads LEADING:3 TRAILING:3 SLIDINGWINDOW:4:20 MINLEN:50" \
			--remove-intermediate-output &
	done
	wait
	echo "所有样本的Kneaddata分析已完成！"
	```

# 宏基因组去冗余
> 对于初学者，推荐从CD-HIT开始，简单易用
> 如果数据量极大且资源有限，CD-HIT是最佳选择
> 如果需要高精度区分近缘物种，选择MMseqs2
> 如果是基因组级别的去冗余，必须使用dRep
## CD-HIT
> 核心算法: 贪心聚类算法
> 主要特点: 经典工具，速度极快，内存占用低
> 适用场景: 大规模基因集去冗余、蛋白质序列聚类
> 速度: ★★★★★（极快）
> 推荐参数: -c 0.95 -n 5 -M 16000 -T 8
1. 步骤1：基因预测（如果contigs尚未预测基因）
	```bash
	##!/bin/bash
	#step1_gene_prediction.sh
	# 使用Prodigal进行基因预测
	input_contigs="assembly_contigs.fasta"
	output_genes="predicted_genes.faa"
	# 预测蛋白质编码基因
	prodigal -i $input_contigs \
		 	-a $output_genes \
		 	-p meta \
		 	-q
	echo "基因预测完成！共预测 $(grep -c '^>' $output_genes) 个基因"
	```
2. 步骤2：CD-HIT聚类去冗余
	```bash
	##!/bin/bash
	# step2_cdhit_derep.sh
	input_genes="predicted_genes.faa"
	output_nr="nonredundant_genes.faa"
	clustering_info="cdhit_clusters.clstr"
	# 设置相似度阈值（通常90-95%用于蛋白质）
	similarity=0.95
	echo "开始CD-HIT去冗余，相似度阈值: $similarity"
	cd-hit -i $input_genes \
	   	-o $output_nr \
	   	-c $similarity \
	   	-n 5 \
	   	-M 16000 \
	   	-T 8 \
	   	-d 0 \
	   	-sc 1
	echo "去冗余完成！"
	echo "输入基因数: $(grep -c '^>' $input_genes)"
	echo "去冗余后基因数: $(grep -c '^>' $output_nr)"
	echo "聚类信息保存在: $clustering_info"
	```
3. 步骤3：分析聚类结果
	```bash
	#!/bin/bash
	# step3_analyze_clusters.sh
	# 统计聚类大小分布
	echo "聚类大小统计："
	cat cdhit_clusters.clstr | awk '/^>/ {if(size>0) print size; size=0} /^[0-9]/ {size++} END {print size}' | \    sort -n | uniq -c | sort -k2n > cluster_size_distribution.txt
	echo "聚类大小分布已保存到 cluster_size_distribution.txt"
	# 提取代表性序列ID
	grep '^>' nonredundant_genes.faa | sed 's/^>//' > representative_ids.txt
	echo "代表性序列ID已保存到 representative_ids.txt"
	```
4. 多样本合并去冗余
	```bash
	#!/bin/bash
	# multi_sample_derep.sh
	# 合并所有样本的基因
	cat sample*_contigs.fasta > all_contigs_combined.fasta
	# 基因预测
	prodigal -i all_contigs_combined.fasta \
		 	-a all_genes.faa \
		 	-p meta \
		 	-q

	   	-o all_genes_nr.faa \
	   	-c 0.9 \          # 90%相似度
	   	-n 5 \
	   	-M 32000 \        # 32GB内存
	   	-T 16 \
	   	-g 1 \            # 精确模式
	   	-d 0
	echo "多样本合并去冗余完成！"
	echo "总基因数: $(grep -c '^>' all_genes.faa)"
	echo "非冗余基因数: $(grep -c '^>' all_genes_nr.faa)"
	```
## MMseqs2
> 核心算法: 敏感序列比对 + 层次聚类
> 主要特点: 灵敏度高，内存效率好，支持多种输出格式
> 适用场景: 复杂微生物群落、近缘物种区分
> 速度: ★★★★☆（较快）
> 推荐参数: --min-seq-id 0.9 -c 0.8 --cov-mode 1
1. 单步完成聚类
	```bash
	#!/bin/bash
	# mmseqs2_derep.sh
	input_genes="predicted_genes.faa"
	output_prefix="mmseqs2_nr"
	tmp_dir="mmseqs2_tmp"
	# 创建数据库
	mmseqs createdb $input_genes ${output_prefix}DB
	# 聚类（90%相似度，80%覆盖度）
	mmseqs cluster ${output_prefix}DB ${output_prefix}Cluster $tmp_dir \
		--min-seq-id 0.9 \
		--cov-mode 1 \
		-c 0.8 \
		--threads 8
	# 提取代表性序列
	mmseqs createtsv ${output_prefix}DB ${output_prefix}DB ${output_prefix}Cluster ${output_prefix}_cluster.tsv
	mmseqs result2repseq ${output_prefix}DB ${output_prefix}Cluster ${output_prefix}DB_rep
	mmseqs result2flat ${output_prefix}DB ${output_prefix}DB ${output_prefix}DB_rep ${output_prefix}.faa
	echo "MMseqs2去冗余完成！"
	echo "输出文件: ${output_prefix}.faa"
	echo "聚类关系: ${output_prefix}_cluster.tsv"
	```
## UCLUST
> 核心算法: 基于序列相似度的快速聚类
> 主要特点: 速度极快，商业化品质，精确度高
> 适用场景: 超大规模数据快速聚类、OTU聚类
> 速度: ★★★★★（极快）
> 注意: 商业软件，需购买许可证
## VSEARCH
> 核心算法: UCLUST算法开源实现
> 主要特点: 开源免费，与UCLUST高度兼容
> 适用场景: 需要兼容UCLUST流程的分析
> 速度: ★★★★☆（较快）
> 推荐参数: --cluster_fast --id 0.97
## dRep
> 核心算法: 多步骤筛选（MASH + ANI + 其他指标）
> 主要特点: 基因组级别去冗余，集成多种评估指标
> 适用场景: MAGs去重、基因组比较
> 速度: ★★★☆☆（中等）
> 推荐参数: -pa 0.9 -sa 0.95
## 去冗余结果评估
1. 评估指标
	```bash
	#!/bin/bash
	# evaluate_derep.sh
	input_faa="predicted_genes.faa"
	nr_faa="nonredundant_genes.faa"
	echo "=== 去冗余效果评估 ==="
	echo "1. 序列数量变化:"
	input_count=$(grep -c '^>' $input_faa)
	nr_count=$(grep -c '^>' $nr_faa)
	reduction_rate=$(echo "scale=2; (1 - $nr_count/$input_count)*100" | bc)
	echo "   输入序列: $input_count"
	echo "   输出序列: $nr_count"
	echo "   减少比例: ${reduction_rate}%"
	echo "2. 序列长度统计:"
	echo "   输入平均长度: $(seqkit stats $input_faa -T | tail -1 | awk '{print $6}')"
	echo "   输出平均长度: $(seqkit stats $nr_faa -T | tail -1 | awk '{print $6}')"
	echo "3. 聚类信息:"
	if [ -f "cdhit_clusters.clstr" ]; then
		cluster_count=$(grep -c '^>' cdhit_clusters.clstr)
		echo "   聚类数量: $cluster_count"
		echo "   平均聚类大小: $(echo "scale=2; $input_count/$cluster_count" | bc)"
	fi
	```
2. 可视化分析
	```python
	#!/usr/bin/env python3
	# plot_cluster_distribution.py
	import matplotlib.pyplot as plt
	import numpy as np
	# 读取聚类大小分布
	sizes = []
	with open('cluster_size_distribution.txt', 'r') as f:
		for line in f:
			count, size = map(int, line.strip().split())
			sizes.extend([size] * count)
	# 绘制分布图
	fig, axes = plt.subplots(1, 2, figsize=(12, 5))

	sorted_sizes = np.sort(sizes)
	cumulative = np.arange(1, len(sorted_sizes)+1) / len(sorted_sizes)
	axes[1].plot(sorted_sizes, cumulative, 'r-', linewidth=2)
	axes[1].set_xlabel('Cluster Size')
	axes[1].set_ylabel('Cumulative Proportion')
	axes[1].set_title('Cumulative Distribution')
	axes[1].grid(True, alpha=0.3)
	plt.tight_layout()
	plt.savefig('cluster_distribution.png', dpi=300, bbox_inches='tight')
	plt.show()
	```

## 基因目录
1. 构建个人基因目录
	```bash
	#!/bin/bash
	# build_gene_catalog.sh
	# 1. 合并所有样本的基因
	cat */predicted_genes.faa > all_samples_genes.faa
	# 2. 去冗余（构建核心基因集）
	cd-hit  -i all_samples_genes.faa \
			-o gene_catalog.faa \
			-c 0.95 \          # 95%相似度
			-n 5 \
			-M 64000 \         # 64GB内存
			-T 32 \
			-g 1 \
			-d 0
	# 3. 为每个基因分配唯一ID
	awk '/^>/ {printf(">GC_%08d\n", ++i); next} {print}' gene_catalog.faa > gene_catalog_final.faa
	# 4. 创建基因信息表
	echo "gene_id,original_id,length,cluster" > gene_catalog_info.csv
	grep '^>' gene_catalog.faa | sed 's/^>//' | awk -F' ' '{print NR","$1","length($0)}' >> gene_catalog_info.csv
	echo "基因目录构建完成！"
	echo "总基因数: $(grep -c '^>' gene_catalog_final.faa)"
	```
2. 基因目录的应用
	```bash
	#!/bin/bash
	# use_gene_catalog.sh
	gene_catalog="gene_catalog_final.faa"
	sample_genes="sample1_genes.faa"
	# 将样本基因比对到基因目录
	# 使用diamond进行快速比对
	diamond makedb --in $gene_catalog -d gene_catalog_db
	diamond blastp  -d gene_catalog_db \
					-q $sample_genes \
					-o sample1_vs_catalog.m8 \
					--max-target-seqs 1 \
					--evalue 1e-5 \
					--threads 8
	# 统计基因覆盖度
	echo "样本基因在目录中的覆盖度:"
	total_genes=$(grep -c '^>' $sample_genes)
	matched_genes=$(cut -f1 sample1_vs_catalog.m8 | sort -u | wc -l)
	coverage=$(echo "scale=2; $matched_genes/$total_genes*100" | bc)
	echo "  总基因数: $total_genes"
	echo "  匹配基因: $matched_genes"
	echo "  覆盖度: ${coverage}%"
	```
